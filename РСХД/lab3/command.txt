### Этап 3 Восстанавливаем кластер

sudo docker stop primary

#Сносим папку
sudo rm -rf /mnt/pgdisk/primary
sudo mkdir -p /mnt/pgdisk/primary
sudo chown -R 999:999 /mnt/pgdisk/primary

sudo docker compose up -d --build primary # билдим заново ибо 


# Делаем backup со standby
sudo docker exec -it primary bash -lc "
rm -rf /tmp/backup &&
mkdir -p /tmp/backup &&
chown -R postgres:postgres /tmp/backup
"

sudo docker exec -it primary bash -lc "
su postgres -c '
PGPASSWORD=123 \
pg_basebackup -h standby \
  -D /tmp/backup \
  -U replicator \
  -Fp -Xs -P -R
'
"

# Вытаскиваем backup на хост 
sudo rm -rf /tmp/primary_backup_copy
sudo mkdir -p /tmp/primary_backup_copy
sudo docker cp primary:/tmp/backup/. /tmp/primary_backup_copy/

#Сносим primary
sudo docker rm -f primary

# Копируем backup на пустую директорию primary
sudo cp -a /tmp/primary_backup_copy/. /mnt/pgdisk/primary/
sudo chown -R 999:999 /mnt/pgdisk/primary

# Билдим primary заново 
sudo docker compose up -d --build primary

# Возвращаем изначальные права 
sudo docker stop standby
sudo docker exec -it primary bash -lc "psql -U postgres -c 'SELECT pg_promote();'"
sudo docker exec -it primary bash -lc "psql -U postgres -c 'SELECT pg_is_in_recovery();'"

# Удаляем standby
sudo docker rm -f standby
